WITH citas_unicas AS (
    SELECT 
        TO_CHAR(c.cithor, 'HH24:MI:SS') AS hora, 
        p.pactid, 
        c.citcar, 
        TRIM(REPLACE(
            COALESCE(p.pacnom, '') || ' ' || 
            COALESCE(p.pacno2, '') || ' ' || 
            COALESCE(p.pacap1, '') || ' ' || 
            COALESCE(p.pacap2, ''), 
        '  ', ' ')) AS nombre, 
        p.pacsex,
        c.citest,
        c.cithor,
        ROW_NUMBER() OVER (PARTITION BY c.citcar ORDER BY c.cithor ASC) AS rn
    FROM cncit c
    LEFT JOIN abpac p ON c.citcar = p.pacide 
    WHERE c.citmed = '08' 
    AND c.citesp = '751'
      AND c.citfci = '18/11/26' 
      AND c.citest IN ('T', 'P')
)
SELECT 
    hora, 
    pactid, 
    citcar, 
    nombre, 
    pacsex, 
    citest
FROM citas_unicas
WHERE rn = 1
ORDER BY cithor ASC;

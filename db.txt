-- =============================================================================

-- =============================================================================

-- -----------------------------------------------------------------------------
-- LLAMADOR_SEDES
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_SEDES (
    ID              NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOMBRE          VARCHAR2(200) NOT NULL,
    ACTIVO          NUMBER(1) DEFAULT 1 NOT NULL,
    FECHA_CREACION  TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT CK_LLAMADOR_SEDES_ACTIVO CHECK (ACTIVO IN (0, 1))
);

COMMENT ON TABLE LLAMADOR_SEDES IS 'Sedes';
COMMENT ON COLUMN LLAMADOR_SEDES.NOMBRE IS 'Nombre comercial de la sede';

CREATE INDEX IDX_LLAMADOR_SEDES_ACTIVO ON LLAMADOR_SEDES (ACTIVO);

-- -----------------------------------------------------------------------------
-- LLAMADOR_SALAS
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_SALAS (
    ID              NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    SEDE_ID         NUMBER NOT NULL,
    NOMBRE          VARCHAR2(200) NOT NULL,
    ACTIVO          NUMBER(1) DEFAULT 1 NOT NULL,
    FECHA_CREACION  TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT FK_LLAMADOR_SALAS_SEDE FOREIGN KEY (SEDE_ID) REFERENCES LLAMADOR_SEDES (ID),
    CONSTRAINT CK_LLAMADOR_SALAS_ACTIVO CHECK (ACTIVO IN (0, 1))
);

COMMENT ON TABLE LLAMADOR_SALAS IS 'Salas o zonas dentro de una sede';

CREATE INDEX IDX_LLAMADOR_SALAS_SEDE ON LLAMADOR_SALAS (SEDE_ID);
CREATE INDEX IDX_LLAMADOR_SALAS_ACTIVO ON LLAMADOR_SALAS (ACTIVO);

-- -----------------------------------------------------------------------------
-- LLAMADOR_CONSULTORIOS
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_CONSULTORIOS (
    ID              NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    SALA_ID         NUMBER NOT NULL,
    NUMERO          VARCHAR2(50) NOT NULL,
    DESCRIPCION     VARCHAR2(500),
    ACTIVO          NUMBER(1) DEFAULT 1 NOT NULL,
    FECHA_CREACION  TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT FK_LLAMADOR_CONSULTORIOS_SALA FOREIGN KEY (SALA_ID) REFERENCES LLAMADOR_SALAS (ID),
    CONSTRAINT CK_LLAMADOR_CONSULTORIOS_ACTIVO CHECK (ACTIVO IN (0, 1)),
    CONSTRAINT UQ_LLAMADOR_CONSULTORIO_SALA_NUMERO UNIQUE (SALA_ID, NUMERO)
);

COMMENT ON TABLE LLAMADOR_CONSULTORIOS IS 'Consultorios donde atienden los médicos';
COMMENT ON COLUMN LLAMADOR_CONSULTORIOS.NUMERO IS 'Número o código mostrado en pantalla y TTS';

CREATE INDEX IDX_LLAMADOR_CONSULTORIOS_SALA ON LLAMADOR_CONSULTORIOS (SALA_ID);

-- -----------------------------------------------------------------------------
-- LLAMADOR_PANTALLAS
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_PANTALLAS (
    ID              NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    SALA_ID         NUMBER NOT NULL,
    NOMBRE          VARCHAR2(200) NOT NULL,
    CODIGO          VARCHAR2(100) NOT NULL,
    ACTIVO          NUMBER(1) DEFAULT 1 NOT NULL,
    FECHA_CREACION  TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT FK_LLAMADOR_PANTALLAS_SALA FOREIGN KEY (SALA_ID) REFERENCES LLAMADOR_SALAS (ID),
    CONSTRAINT CK_LLAMADOR_PANTALLAS_ACTIVO CHECK (ACTIVO IN (0, 1)),
    CONSTRAINT UQ_LLAMADOR_PANTALLAS_CODIGO UNIQUE (CODIGO),
    CONSTRAINT UQ_LLAMADOR_PANTALLAS_SALA UNIQUE (SALA_ID)
);

COMMENT ON TABLE LLAMADOR_PANTALLAS IS 'Televisor/computador que muestra llamadas de la sala';
COMMENT ON COLUMN LLAMADOR_PANTALLAS.CODIGO IS 'Identificador público usado en URL /pantalla/{codigo}';

CREATE INDEX IDX_LLAMADOR_PANTALLAS_SALA ON LLAMADOR_PANTALLAS (SALA_ID);

-- -----------------------------------------------------------------------------
-- LLAMADOR_MEDICOS
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_MEDICOS (
    ID              NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOMBRE          VARCHAR2(200) NOT NULL,
    ACTIVO          NUMBER(1) DEFAULT 1 NOT NULL,
    FECHA_CREACION  TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT CK_LLAMADOR_MEDICOS_ACTIVO CHECK (ACTIVO IN (0, 1))
);

COMMENT ON TABLE LLAMADOR_MEDICOS IS 'Profesionales que realizan llamadas';

-- -----------------------------------------------------------------------------
-- LLAMADOR_USUARIOS
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_USUARIOS (
    ID              NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    USERNAME        VARCHAR2(100) NOT NULL,
    PASSWORD_HASH   VARCHAR2(255) NOT NULL,
    ROL             VARCHAR2(20) NOT NULL,
    MEDICO_ID       NUMBER,
    ACTIVO          NUMBER(1) DEFAULT 1 NOT NULL,
    FECHA_CREACION  TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    CONSTRAINT FK_LLAMADOR_USUARIOS_MEDICO FOREIGN KEY (MEDICO_ID) REFERENCES LLAMADOR_MEDICOS (ID),
    CONSTRAINT CK_LLAMADOR_USUARIOS_ROL CHECK (ROL IN ('ADMIN', 'GESTOR', 'MEDICO', 'PANTALLA')),
    CONSTRAINT CK_LLAMADOR_USUARIOS_ACTIVO CHECK (ACTIVO IN (0, 1)),
    CONSTRAINT UQ_LLAMADOR_USUARIOS_USERNAME UNIQUE (USERNAME)
);

COMMENT ON TABLE LLAMADOR_USUARIOS IS 'Credenciales del sistema';
COMMENT ON COLUMN LLAMADOR_USUARIOS.ROL IS 'ADMIN, GESTOR, MEDICO o PANTALLA';

-- -----------------------------------------------------------------------------
-- LLAMADOR_LLAMADAS
-- ESTADO: ACTIVA = llamada actual en pantalla; FINALIZADA = solo estadísticas
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_LLAMADAS (
    ID                  NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    PANTALLA_ID         NUMBER NOT NULL,
    CONSULTORIO_ID      NUMBER NOT NULL,
    MEDICO_ID           NUMBER,
    PACIENTE_NOMBRE     VARCHAR2(300) NOT NULL,
    ESTADO              VARCHAR2(20) NOT NULL,
    FECHA_LLAMADA       TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    FECHA_EXPIRACION    TIMESTAMP NOT NULL,
    FECHA_FINALIZACION  TIMESTAMP,
    CONSTRAINT FK_LLAMADOR_LLAMADAS_PANTALLA FOREIGN KEY (PANTALLA_ID) REFERENCES LLAMADOR_PANTALLAS (ID),
    CONSTRAINT FK_LLAMADOR_LLAMADAS_CONSULTORIO FOREIGN KEY (CONSULTORIO_ID) REFERENCES LLAMADOR_CONSULTORIOS (ID),
    CONSTRAINT FK_LLAMADOR_LLAMADAS_MEDICO FOREIGN KEY (MEDICO_ID) REFERENCES LLAMADOR_MEDICOS (ID),
    CONSTRAINT CK_LLAMADOR_LLAMADAS_ESTADO CHECK (ESTADO IN ('ACTIVA', 'FINALIZADA'))
);

COMMENT ON TABLE LLAMADOR_LLAMADAS IS 'Registro de llamadas; solo ACTIVA no expirada es la llamada vigente';
COMMENT ON COLUMN LLAMADOR_LLAMADAS.FECHA_EXPIRACION IS 'Tras esta fecha la pantalla debe volver a espera';

CREATE INDEX IDX_LLAMADOR_LLAMADAS_PANTALLA_ESTADO ON LLAMADOR_LLAMADAS (PANTALLA_ID, ESTADO);
CREATE INDEX IDX_LLAMADOR_LLAMADAS_FECHA ON LLAMADOR_LLAMADAS (FECHA_LLAMADA);
CREATE INDEX IDX_LLAMADOR_LLAMADAS_EXPIRACION ON LLAMADOR_LLAMADAS (FECHA_EXPIRACION);

-- -----------------------------------------------------------------------------
-- LLAMADOR_AUDITORIA
-- -----------------------------------------------------------------------------
CREATE TABLE LLAMADOR_AUDITORIA (
    ID              NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    USUARIO_ID      NUMBER,
    NOMBRE_USUARIO  VARCHAR2(200) NOT NULL,
    ROL_USUARIO     VARCHAR2(20) NOT NULL,
    MODULO          VARCHAR2(30) NOT NULL,
    ACCION          VARCHAR2(50) NOT NULL,
    DETALLES        VARCHAR2(4000),
    FECHA           TIMESTAMP NOT NULL,
    CONSTRAINT FK_LLAMADOR_AUDITORIA_USUARIO FOREIGN KEY (USUARIO_ID) REFERENCES LLAMADOR_USUARIOS (ID)
);

COMMENT ON TABLE LLAMADOR_AUDITORIA IS 'Trazabilidad de acciones críticas del sistema';
COMMENT ON COLUMN LLAMADOR_AUDITORIA.FECHA IS 'Hora local America/Bogota';

CREATE INDEX IDX_LLAMADOR_AUDITORIA_FECHA ON LLAMADOR_AUDITORIA (FECHA);
CREATE INDEX IDX_LLAMADOR_AUDITORIA_ROL ON LLAMADOR_AUDITORIA (ROL_USUARIO);
